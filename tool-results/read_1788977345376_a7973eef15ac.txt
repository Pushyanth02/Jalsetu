    74→export function EventDetailView() {

    76→  const { toast } = useToast();
    77→  const qc = useQueryClient();
    78→
    79→  const detailQ = useQuery({
    80→    queryKey: ["event", eventId],
    81→    queryFn: () => apiGet<EventDetailResponse>(`/api/events/${eventId}`).then((r) => r.data),
    82→    enabled: !!eventId,
    83→  });
    84→
    85→  const invalidate = () => {
    86→    qc.invalidateQueries({ queryKey: ["event", eventId] });
    87→    qc.invalidateQueries({ queryKey: ["events"] });
    88→    qc.invalidateQueries({ queryKey: ["overview"] });
    89→    qc.invalidateQueries({ queryKey: ["risk"] });
    90→  };
    91→
    92→  const reassess = useMutation({
    93→    mutationFn: () => apiPost(`/api/events/${eventId}/reassess`, { useProvider: true }, "ANALYST"),
    94→    onSuccess: (r) => {
    95→      invalidate();

    98→    onError: (e) => toast({ title: "Reassessment failed", description: (e as ApiClientError).message, variant: "destructive" }),
    99→  });

   101→  if (!eventId) {
   102→    return (

   105→          <Button variant="outline" size="sm" onClick={() => navigate("command")} className="mt-2">← Command Center</Button>
   106→        } />
   107→      </div>

   270→function OverviewTab({ d }: { d: D }) {
   271→  const { event: e, context } = d;
   272→  const nearbyAssets = context.assets
   273→    .map((a) => ({ ...a, d: Math.round(hav({ lat: e.lat, lng: e.lng }, a)) }))
   274→    .filter((a) => a.d <= 600)
   275→    .sort((a, b) => a.d - b.d)
   276→    .slice(0, 6);
   277→  const nearIncidents = context.historicalIncidents
   278→    .map((i) => ({ ...i, d: Math.round(hav({ lat: e.lat, lng: e.lng }, i)) }))
   279→    .filter((i) => i.d <= 300)
   280→    .sort((a, b) => b.occurredOn.localeCompare(a.occurredOn))
   281→    .slice(0, 5);
   282→  const nearMaint = context.maintenance
   283→    .filter((m) => m.asset && nearbyAssets.some((a) => a.code === m.asset?.code))
   284→    .slice(0, 4);
   285→
   286→  return (
   287→    <>
   288→      <Panel title={`citizen reports · ${e.reportCount}`}>
   289→        <ul className="divide-y divide-border/60">
   290→          {e.reports.map((r) => (
   291→            <li key={r.id} className="py-2.5 first:pt-0 last:pb-0">
   292→              <div className="flex items-center gap-2 flex-wrap">
   293→                <span className="data-mono text-[0.65rem] text-water/90">{r.publicRef}</span>
   294→                <span className="micro-label !text-[0.52rem]">{r.channel.toLowerCase()}</span>
   295→                {r.isDuplicate && <span className="micro-label !text-[0.52rem] text-sev-moderate/80">merged as duplicate</span>}
   296→                <TimeAgo iso={r.submittedAt} />
   297→              </div>
   298→              <p className="mt-1 text-[0.82rem] text-foreground/90 leading-relaxed">{r.description}</p>
   299→              {r.classification && (
   300→                <div className="mt-1.5 flex items-center gap-2 flex-wrap text-[0.65rem] text-muted-foreground">
   301→                  <span>assessed <span className={cn("font-medium", sevClass(r.classification.severity))}>{r.classification.severity.toLowerCase()}</span></span>
   302→                  <span className="data-mono">conf {Math.round(r.classification.confidence * 100)}%</span>
   303→                  {r.classificationProvider && <ProviderChip provider={r.classificationProvider} />}
   304→                </div>
   305→              )}
   306→            </li>
   307→          ))}
   308→        </ul>
   309→      </Panel>
   310→
   311→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   312→        <Panel title={`infrastructure context · within 600m`}>
   313→          {nearbyAssets.length === 0 ? (
   314→            <EmptyState title="No assets recorded nearby" hint="Asset registry covers pilot jurisdictions only (synthetic)." />
   315→          ) : (
   316→            <ul className="space-y-2">
   317→              {nearbyAssets.map((a) => (
   318→                <li key={a.code} className="flex items-center gap-2.5 text-xs">
   319→                  <span className="data-mono text-[0.62rem] text-muted-foreground w-12 shrink-0">{a.d}m</span>
   320→                  <span className={cn("size-2 rounded-full shrink-0", a.conditionScore < 45 ? "bg-sev-high" : a.conditionScore < 65 ? "bg-sev-moderate" : "bg-verified")} aria-hidden />
   321→                  <span className="flex-1 min-w-0">
   322→                    <span className="block truncate">{a.name}</span>
   323→                    <span className="micro-label !text-[0.5rem]">{a.kind.toLowerCase().replace(/_/g, " ")} · {a.agencyCode} · condition {a.conditionScore}</span>
   324→                  </span>
   325→                </li>
   326→              ))}
   327→            </ul>
   328→          )}
   329→        </Panel>
   330→
   331→        <Panel title="historical incidents · within 300m">
   332→          {nearIncidents.length === 0 ? (
   333→            <EmptyState title="No historical incidents nearby" hint="Synthetic archive covers the pilot jurisdictions." />
   334→          ) : (
   335→            <ul className="space-y-2">
   336→              {nearIncidents.map((i) => (
   337→                <li key={i.id} className="flex items-baseline gap-2.5 text-xs">
   338→                  <span className="data-mono text-[0.62rem] text-muted-foreground shrink-0 w-20">{new Date(i.occurredOn).toLocaleDateString("en-IN", { month: "short", year: "numeric" })}</span>
   339→                  <span className={cn("font-medium shrink-0", sevClass(i.severity))}>{i.severity.toLowerCase()}</span>
   340→                  <span className="text-muted-foreground truncate">{i.waterDepthCm ?? "?"}cm · {i.durationHours?.toFixed(0) ?? "?"}h · {i.reportedVia.toLowerCase()}</span>
   341→                </li>
   342→              ))}
   343→            </ul>
   344→          )}
   345→        </Panel>
   346→      </div>
   347→
   348→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   349→        <Panel title="maintenance nearby">
   350→          {nearMaint.length === 0 ? (
   351→            <EmptyState title="No maintenance records for nearby assets" />
   352→          ) : (
   353→            <ul className="space-y-2">
   354→              {nearMaint.map((m, i) => (
   355→                <li key={i} className="flex items-baseline gap-2.5 text-xs">
   356→                  <span className="data-mono text-[0.62rem] text-muted-foreground w-20 shrink-0">
   357→                    {m.performedAt ? new Date(m.performedAt).toLocaleDateString("en-IN", { month: "short", year: "numeric" }) : "scheduled"}
   358→                  </span>
   359→                  <span className="flex-1 truncate">{m.asset?.name ?? "asset"} · {m.kind.toLowerCase()}</span>
   360→                  <span className={cn("micro-label !text-[0.5rem]", m.status === "COMPLETED" ? "text-verified" : "text-water")}>{m.status.toLowerCase()}</span>
   361→                </li>
   362→              ))}
   363→            </ul>
   364→          )}
   365→        </Panel>
   366→
   367→        <Panel title={<span className="flex items-center gap-1.5"><Umbrella className="size-3 text-water" aria-hidden /> rainfall · 72h (synthetic)</span>}>
   368→          <ul className="space-y-1.5 text-xs">
   369→            {context.rainfallSeries.slice(-12).map((s, i) => (
   370→              <li key={i} className="flex items-center gap-2">
   371→                <span className="data-mono text-[0.6rem] text-muted-foreground w-11 shrink-0">{new Date(s.t).toLocaleTimeString("en-IN", { hour: "2-digit", hour12: false })}</span>
   372→                <span className="h-1.5 rounded-full bg-water/70" style={{ width: `${Math.min(100, s.mm * 3)}%` }} />
   373→                <span className="data-mono text-[0.62rem]">{s.mm}mm</span>
   374→              </li>
   375→            ))}
   376→          </ul>
   377→        </Panel>
   378→      </div>
   379→    </>
   380→  );
   381→}

  1005→function hav(a: { lat: number; lng: number }, b: { lat: number; lng: number }): number {
  1006→  const R = 6371000;
  1007→  const rad = (x: number) => (x * Math.PI) / 180;
  1008→  const dLat = rad(b.lat - a.lat);
  1009→  const dLng = rad(b.lng - a.lng);
  1010→  const s = Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLng / 2) ** 2;
  1011→  return 2 * R * Math.asin(Math.sqrt(s));
  1012→}

  1014→async function fileToDataUrl(file: File, maxDim: number, quality: number): Promise<string> {
  1015→  const bitmap = await createImageBitmap(file);
  1016→  const scale = Math.min(1, maxDim / Math.max(bitmap.width, bitmap.height));
  1017→  const canvas = document.createElement("canvas");
  1018→  canvas.width = Math.round(bitmap.width * scale);
  1019→  canvas.height = Math.round(bitmap.height * scale);
  1020→  const ctx = canvas.getContext("2d");
  1021→  if (!ctx) throw new Error("Canvas unavailable");
  1022→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  1023→  bitmap.close();
  1024→  return canvas.toDataURL("image/jpeg", quality);
  1025→}
