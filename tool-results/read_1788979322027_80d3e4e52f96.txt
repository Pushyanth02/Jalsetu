   348→   348→async function generateWeather(rng: Rng, stormPeak: number): Promise<number> {
   349→   349→  let count = 0;
   350→   350→  const slots: number[] = [];
   351→   351→  for (let t = Date.now() - 96 * HOUR; t <= Date.now(); t += 3 * HOUR) slots.push(t);
   352→   352→  for (const s of STATIONS) {
   353→   353→    const stationRng = new Rng(hashSeed(s.code));
   354→   354→    for (const t of slots) {
   355→   355→      const dt = (t - stormPeak) / HOUR;
   356→   356→      const storm = 26 * s.amplitude * Math.exp(-(dt * dt) / (2 * 10 * 10));
   357→   357→      const diurnal = 3.5 * s.amplitude * Math.max(0, Math.sin(((t / HOUR) % 24) / 24 * 2 * Math.PI - Math.PI / 2));
   358→   358→      const noise = stationRng.float(0, 2.2);
   359→   359→      const rainfall = Math.max(0, +(storm + diurnal * 0.3 + noise).toFixed(1));
   360→   360→      await db.weatherObservation.create({
   361→   361→        data: {
   362→   362→          stationCode: s.code,
   363→   363→          stationName: s.name,
   364→   364→          lat: s.lat,
   365→   365→          lng: s.lng,
   366→   366→          observedAt: new Date(t),
   367→   367→          rainfallMm: rainfall,
   368→   368→          source: "SYNTHETIC_DEMO",
   369→   369→        },
   370→   370→      });
   371→   371→      count += 1;
   372→   372→    }
   373→   373→  }
   374→   374→  return count;
   375→   375→}

   775→   775→function nearestHotspot(p: { lat: number; lng: number }): HotspotDef | null {
   776→   776→  let best: HotspotDef | null = null;
   777→   777→  let bestD = Infinity;
   778→   778→  for (const h of HOTSPOTS) {
   779→   779→    const d = distanceM(p, { lat: h.lat, lng: h.lng });
   780→   780→    if (d < bestD) {
   781→   781→      bestD = d;
   782→   782→      best = h;
   783→   783→    }
   784→   784→  }
   785→   785→  return best;
   786→   786→}
