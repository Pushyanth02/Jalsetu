   116→   116→// Downscale an image file to at most maxDim px and return a jpeg data URL.
   117→   117→async function fileToDataUrl(file: File, maxDim: number, quality: number): Promise<string> {
   118→   118→  const bitmap = await createImageBitmap(file);
   119→   119→  const scale = Math.min(1, maxDim / Math.max(bitmap.width, bitmap.height));
   120→   120→  const canvas = document.createElement("canvas");
   121→   121→  canvas.width = Math.round(bitmap.width * scale);
   122→   122→  canvas.height = Math.round(bitmap.height * scale);
   123→   123→  const ctx = canvas.getContext("2d");
   124→   124→  if (!ctx) throw new Error("Canvas unavailable");
   125→   125→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
   126→   126→  bitmap.close();
   127→   127→  return canvas.toDataURL("image/jpeg", quality);
   128→   128→}

   260→   260→function ReportWizard() {
   261→   261→  const { toast } = useToast();
   262→   262→  const qc = useQueryClient();
   263→   263→
   264→   264→  const [step, setStep] = useState(0);

   272→   272→  const [gpsState, setGpsState] = useState<"idle" | "locating" | "ok" | "error">("idle");
   273→   273→  const [gpsMsg, setGpsMsg] = useState("");
   274→   274→
   275→   275→  // step 2: issue
   276→   276→  const [category, setCategory] = useState("WATERLOGGING");
   277→   277→  const [severity, setSeverity] = useState("MEDIUM");
   278→   278→  const [description, setDescription] = useState("");
   279→   279→
   280→   280→  // step 3: evidence
   281→   281→  const [photo, setPhoto] = useState<Photo | null>(null);
   282→   282→  const [photoBusy, setPhotoBusy] = useState(false);
   283→   283→  const [photoErr, setPhotoErr] = useState<string | null>(null);
   284→   284→  const [phone, setPhone] = useState("");
   285→   285→  const [consent, setConsent] = useState(false);
   286→   286→
   287→   287→  // step 4: result
   288→   288→  const [outcome, setOutcome] = useState<ReportOutcome | null>(null);
   289→   289→
   290→   290→  const jurisdictionsQ = useQuery({
   291→   291→    queryKey: ["jurisdictions"],
   292→   292→    queryFn: () => apiGet<JurisdictionResponse[]>("/api/jurisdictions").then((r) => r.data),
   293→   293→    staleTime: 600_000,
   294→   294→  });
   295→   295→  const jurisdictions = jurisdictionsQ.data ?? [];
   296→   296→
   297→   297→  const latNum = Number(latText);
   298→   298→  const lngNum = Number(lngText);
   299→   299→  const latValid = latText.trim() !== "" && Number.isFinite(latNum) && latNum >= 28.3 && latNum <= 28.9;
   300→   300→  const lngValid = lngText.trim() !== "" && Number.isFinite(lngNum) && lngNum >= 76.8 && lngNum <= 77.6;
   301→   301→  const coordsValid = latValid && lngValid;
   302→   302→
   303→   303→  const descTrim = description.trim();
   304→   304→  const descValid = descTrim.length >= 12 && descTrim.length <= 600;
   305→   305→  const phoneTrim = phone.trim();
   306→   306→  const phoneValid = phoneTrim === "" || PHONE_RE.test(phoneTrim);
   307→   307→
   308→   308→  const stepValid = [coordsValid, descValid, phoneValid && consent, true][step];
   309→   309→
   310→   310→  const onJurisdiction = (v: string) => {
   311→   311→    setJurisdictionId(v);
   312→   312→    if (v !== "none") {
   313→   313→      const j = jurisdictions.find((x) => x.id === v);
   314→   314→      if (j) {
   315→   315→        setLatText(j.centroid.lat.toFixed(5));
   316→   316→        setLngText(j.centroid.lng.toFixed(5));
   317→   317→      }
   318→   318→    }
   319→   319→  };
   320→   320→
   321→   321→  const onGps = () => {
   322→   322→    if (typeof navigator === "undefined" || !("geolocation" in navigator)) {
   323→   323→      setGpsState("error");
   324→   324→      setGpsMsg("Geolocation is not available in this browser. Enter coordinates manually.");
   325→   325→      return;
   326→   326→    }
   327→   327→    setGpsState("locating");
   328→   328→    setGpsMsg("");
   329→   329→    navigator.geolocation.getCurrentPosition(
   330→   330→      (pos) => {
   331→   331→        setLatText(pos.coords.latitude.toFixed(5));
   332→   332→        setLngText(pos.coords.longitude.toFixed(5));
   333→   333→        setGpsState("ok");
   334→   334→        setGpsMsg("Device coordinates captured. They must fall inside the pilot window (lat 28.3 to 28.9, lng 76.8 to 77.6) to submit.");
   335→   335→      },
   336→   336→      () => {
   337→   337→        setGpsState("error");
   338→   338→        setGpsMsg("Location permission denied or unavailable. Enter coordinates manually or pick a pilot area.");
   339→   339→      },
   340→   340→      { enableHighAccuracy: true, timeout: 8000, maximumAge: 30_000 }
   341→   341→    );
   342→   342→  };
   343→   343→
   344→   344→  const fileRef = useRef<HTMLInputElement>(null);
   345→   345→  const onPhotoFile = async (file: File) => {
   346→   346→    setPhotoBusy(true);
   347→   347→    setPhotoErr(null);
   348→   348→    try {
   349→   349→      if (!file.type.startsWith("image/")) throw new Error("Please select an image file.");
   350→   350→      const dataUrl = await fileToDataUrl(file, 1024, 0.7);
   351→   351→      setPhoto({ dataUrl, name: file.name });
   352→   352→    } catch (e) {
   353→   353→      setPhotoErr((e as Error).message);
   354→   354→    } finally {
   355→   355→      setPhotoBusy(false);
   356→   356→      if (fileRef.current) fileRef.current.value = "";
   357→   357→    }
   358→   358→  };
   359→   359→
   360→   360→  const submit = useMutation({
   361→   361→    mutationFn: () =>
   362→   362→      apiPost<ReportOutcome>(
   363→   363→        "/api/reports",
   364→   364→        {
   365→   365→          description: descTrim,
   366→   366→          category,
   367→   367→          severityReported: severity,
   368→   368→          lat: latNum,
   369→   369→          lng: lngNum,
   370→   370→          addressText: addressText.trim() || undefined,
   371→   371→          channel: "WEB",
   372→   372→          reporterPhone: phoneTrim || undefined,
   373→   373→          consentGiven: true,
   374→   374→          photoDataUrl: photo?.dataUrl,
   375→   375→        },
   376→   376→        "CITIZEN"
   377→   377→      ),
   378→   378→    onSuccess: (r) => {
   379→   379→      setOutcome(r.data);
   380→   380→      setStep(4);
   381→   381→      qc.invalidateQueries({ queryKey: ["events"] });
   382→   382→      qc.invalidateQueries({ queryKey: ["overview"] });
   383→   383→      toast({ title: "Report submitted", description: `Tracking reference ${r.data.publicRef}. Keep it to follow the outcome.` });
   384→   384→    },
   385→   385→    onError: (e) => {
   386→   386→      toast({ title: "Submission failed", description: (e as ApiClientError).message, variant: "destructive" });
   387→   387→    },
   388→   388→  });
   389→   389→
   390→   390→  const goTo = (n: number) => {
   391→   391→    setAttempted(false);
   392→   392→    setStep(n);
   393→   393→  };
   394→   394→
   395→   395→  const next = () => {
   396→   396→    if (!stepValid) {
   397→   397→      setAttempted(true);
   398→   398→      return;
   399→   399→    }
   400→   400→    if (step === 3) {
   401→   401→      submit.mutate();
   402→   402→      return;
   403→   403→    }
   404→   404→    goTo(Math.min(3, step + 1));
   405→   405→  };
   406→   406→
   407→   407→  const reset = () => {
   408→   408→    setStep(0);
   409→   409→    setAttempted(false);
   410→   410→    setLatText("");
   411→   411→    setLngText("");
   412→   412→    setJurisdictionId("none");
   413→   413→    setAddressText("");
   414→   414→    setGpsState("idle");
