    84→
    85→  const detailQ = useQuery({
    86→    queryKey: ["event", eventId],
    87→    queryFn: () => apiGet<EventDetailResponse>(`/api/events/${eventId}`).then((r) => r.data),
    88→    enabled: !!eventId,
    89→  });
    90→
    91→  const invalidate = () => {
    92→    qc.invalidateQueries({ queryKey: ["event", eventId] });

   108→    return (

   112→        } />
   113→      </div>
   114→    );

   243→/** Tab trigger with an OriginKit-style sliding white pill (layout animation). */
   244→function DossierTab({ value, current, children }: { value: string; current: string; children: React.ReactNode }) {
   245→  const reduce = useReducedMotion();
   246→  const active = value === current;
   247→  return (
   248→    <TabsTrigger
   249→      value={value}
   250→      className={cn(
   251→        "relative rounded-md text-xs font-medium text-slate-500 hover:text-slate-700 data-[state=active]:text-slate-900",
   252→        reduce
   253→          ? "data-[state=active]:bg-white data-[state=active]:shadow-sm"
   254→          : "data-[state=active]:bg-transparent data-[state=active]:shadow-none"
   255→      )}
   256→    >
   257→      {active && !reduce && (
   258→        <motion.span
   259→          layoutId="dossier-tab-indicator"
   260→          className="absolute inset-0 rounded-md bg-white shadow-sm"
   261→          aria-hidden
   262→          transition={{ type: "spring", stiffness: 420, damping: 34 }}
   263→        />
   264→      )}
   265→      <span className="relative z-10">{children}</span>
   266→    </TabsTrigger>
   267→  );
   268→}

   309→function OverviewTab({ d }: { d: D }) {
   310→  const { event: e, context } = d;
   311→  const nearbyAssets = context.assets
   312→    .map((a) => ({ ...a, d: Math.round(hav({ lat: e.lat, lng: e.lng }, a)) }))
   313→    .filter((a) => a.d <= 600)
   314→    .sort((a, b) => a.d - b.d)
   315→    .slice(0, 6);
   316→  const nearIncidents = context.historicalIncidents
   317→    .map((i) => ({ ...i, d: Math.round(hav({ lat: e.lat, lng: e.lng }, i)) }))
   318→    .filter((i) => i.d <= 300)
   319→    .sort((a, b) => b.occurredOn.localeCompare(a.occurredOn))
   320→    .slice(0, 5);
   321→  const nearMaint = context.maintenance
   322→    .filter((m) => m.asset && nearbyAssets.some((a) => a.code === m.asset?.code))
   323→    .slice(0, 4);
   324→
   325→  return (
   326→    <>
   327→      <Reveal>
   328→        <Panel title={`citizen reports · ${e.reportCount}`} icon={<MessageSquare />}>
   329→          <Stagger className="divide-y divide-border/60">
   330→            {e.reports.map((r) => (
   331→              <StaggerItem key={r.id} className="py-2.5 first:pt-0 last:pb-0">
   332→                <div className="flex items-center gap-2 flex-wrap">
   333→                  <span className="data-mono text-[0.65rem] text-water">{r.publicRef}</span>

   356→              <EmptyState title="No assets recorded nearby" hint="Asset registry covers pilot jurisdictions only (synthetic)." />
   357→            ) : (
   358→              <ul className="space-y-2">
   359→                {nearbyAssets.map((a) => (
   360→                  <li key={a.code} className="flex items-center gap-2.5 text-xs">
   361→                    <span className="data-mono text-[0.62rem] text-muted-foreground w-12 shrink-0">{a.d}m</span>

   363→                    <span className="flex-1 min-w-0">
   364→                      <span className="block truncate text-slate-700">{a.name}</span>
