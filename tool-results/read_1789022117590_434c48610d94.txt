   226→function DeltaPill({ delta }: { delta: number }) {
   227→  const up = delta > 0;
   228→  const flat = delta === 0;
   229→  return (
   230→    <span
   231→      className={cn(
   232→        "inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[0.6rem] font-semibold",
   233→        flat ? "bg-slate-100 text-slate-500" : up ? "bg-red-50 text-sev-critical" : "bg-emerald-50 text-verified"
   234→      )}
   235→      title="new events in last 24h vs prior 24h (computed from pilot data)"
   236→    >
   237→      {flat ? <Minus className="size-3" aria-hidden /> : up ? <TrendingUp className="size-3" aria-hidden /> : <TrendingDown className="size-3" aria-hidden />}
   238→      {delta > 0 ? "+" : ""}{delta}%
   239→    </span>
   240→  );
   241→}

   243→function eventDelta(events?: EventSummary[]): number | null {
   244→  if (!events?.length) return null;
   245→  const now = Date.now();
   246→  const day = 86_400_000;
   247→  const recent = events.filter((e) => new Date(e.firstReportedAt).getTime() > now - day).length;
   248→  const prior = events.filter((e) => {
   249→    const t = new Date(e.firstReportedAt).getTime();
   250→    return t <= now - day && t > now - 2 * day;
   251→  }).length;
   252→  if (prior === 0) return recent === 0 ? 0 : 100;
   253→  return Math.round(((recent - prior) / prior) * 100);
   254→}
   255→
   256→function bandDelta(events: EventSummary[] | undefined, kind: "HIGH" | "UNRESOLVED"): number | null {
   257→  if (!events?.length) return null;
   258→  const now = Date.now();
   259→  const day = 86_400_000;
   260→  const inBand = (e: EventSummary) => (kind === "HIGH" ? e.riskBand === "HIGH" || e.riskBand === "CRITICAL" : e.status === "DETECTED" || e.status === "TRIAGED");
   261→  const recent = events.filter((e) => inBand(e) && new Date(e.firstReportedAt).getTime() > now - day).length;
   262→  const prior = events.filter((e) => inBand(e) && new Date(e.firstReportedAt).getTime() <= now - day && new Date(e.firstReportedAt).getTime() > now - 2 * day).length;
   263→  if (prior === 0) return recent === 0 ? 0 : 100;
   264→  return Math.round(((recent - prior) / prior) * 100);
   265→}

   643→
   644→function rainfallSeries(w?: WeatherResponse) {
   645→  if (!w) return [];
   646→  const bySlot = new Map<string, { t: number; mm: number; n: number }>();
   647→  for (const o of w.series) {
   648→    const key = o.t.slice(0, 16);
   649→    const slot = bySlot.get(key) ?? { t: new Date(o.t).getTime(), mm: 0, n: 0 };
   650→    slot.mm += o.mm;
   651→    slot.n += 1;
   652→    bySlot.set(key, slot);
   653→  }
   654→  return [...bySlot.values()]
   655→    .sort((a, b) => a.t - b.t)
   656→    .map((s) => ({
   657→      label: new Date(s.t).toLocaleTimeString("en-IN", { hour: "2-digit", hour12: false }),
   658→      mm: +(s.mm / s.n).toFixed(1),
   659→    }));
   660→}
