   348→async function generateWeather(rng: Rng, stormPeak: number): Promise<number> {
   349→  let count = 0;
   350→  const slots: number[] = [];
   351→  for (let t = Date.now() - 96 * HOUR; t <= Date.now(); t += 3 * HOUR) slots.push(t);
   352→  for (const s of STATIONS) {
   353→    const stationRng = new Rng(hashSeed(s.code));
   354→    for (const t of slots) {
   355→      const dt = (t - stormPeak) / HOUR;
   356→      const storm = 26 * s.amplitude * Math.exp(-(dt * dt) / (2 * 10 * 10));
   357→      const diurnal = 3.5 * s.amplitude * Math.max(0, Math.sin(((t / HOUR) % 24) / 24 * 2 * Math.PI - Math.PI / 2));
   358→      const noise = stationRng.float(0, 2.2);
   359→      const rainfall = Math.max(0, +(storm + diurnal * 0.3 + noise).toFixed(1));
   360→      await db.weatherObservation.create({
   361→        data: {
   362→          stationCode: s.code,
   363→          stationName: s.name,
   364→          lat: s.lat,
   365→          lng: s.lng,
   366→          observedAt: new Date(t),
   367→          rainfallMm: rainfall,
   368→          source: "SYNTHETIC_DEMO",
   369→        },
   370→      });
   371→      count += 1;
   372→    }
   373→  }
   374→  return count;
   375→}

   377→async function generateHistoricalIncidents(rng: Rng) {
   378→  for (const h of HOTSPOTS) {
   379→    const n = rng.int(4, 7);
   380→    for (let i = 0; i < n; i++) {
   381→      const year = rng.int(new Date().getFullYear() - 5, new Date().getFullYear() - 1);
   382→      const month = rng.int(6, 9);
   383→      const day = rng.int(1, 28);
   384→      const p = jitter({ lat: h.lat, lng: h.lng }, 90, () => rng.float(0, 1));
   385→      await db.historicalIncident.create({
   386→        data: {
   387→          jurisdictionId: jurisdictionIdFor(h.jurisdiction)!,
   388→          lat: p.lat,
   389→          lng: p.lng,
   390→          occurredOn: new Date(Date.UTC(year, month - 1, day, 9, 30)),
   391→          severity: rng.weighted([
   392→            ["LOW", 1],
   393→            ["MODERATE", 2],
   394→            ["HIGH", 3],
   395→            ["CRITICAL", 2],
   396→          ] as [string, number][]),
   397→          durationHours: rng.float(2, 11),
   398→          waterDepthCm: rng.int(15, 80),
   399→          reportedVia: rng.weighted([
   400→            ["MEDIA", 2],
   401→            ["AGENCY", 1],
   402→            ["HOTLINE", 1],
   403→          ] as [string, number][]),
   404→          source: "SYNTHETIC_DEMO",
   405→        },
   406→      });
   407→    }
   408→  }
   409→  // background incidents away from hotspots
   410→  const noise = new Rng(hashSeed("incidents-noise"));
   411→  let placed = 0;
   412→  let guard = 0;
   413→  while (placed < 26 && guard < 400) {
   414→    guard += 1;
   415→    const p = {
   416→      lat: noise.float(28.61, 28.71),
   417→      lng: noise.float(77.185, 77.26),
   418→    };
   419→    if (!insideAnyJurisdiction(p)) continue;
   420→    if (nearestHotspot(p) && distanceM(p, nearestHotspot(p)!) < 350) continue;
   421→    const j = JURISDICTIONS.find((x) => pointInPolygon(p, [ringOf(x)]))!;
   422→    await db.historicalIncident.create({
   423→      data: {
   424→        jurisdictionId: jurisdictionIdFor(j.code)!,
   425→        lat: p.lat,
   426→        lng: p.lng,
   427→        occurredOn: new Date(Date.UTC(noise.int(new Date().getFullYear() - 5, new Date().getFullYear() - 1), noise.int(6, 9), noise.int(1, 28), 9, 30)),
   428→        severity: noise.weighted([
   429→          ["LOW", 3],
   430→          ["MODERATE", 2],
   431→          ["HIGH", 1],
   432→        ] as [string, number][]),
   433→        durationHours: noise.float(1, 5),
   434→        waterDepthCm: noise.int(5, 30),
   435→        reportedVia: "HOTLINE",
   436→        source: "SYNTHETIC_DEMO",
   437→      },
   438→    });
   439→    placed += 1;
   440→  }
   441→}

   450→function buildReportTimeline(rng: Rng): { items: TimelineItem[] } {
   451→  const items: TimelineItem[] = [];
   452→  const HOUR_ = HOUR;
   453→
   454→  const push = (
   455→    at: Date,
   456→    h: HotspotDef,
   457→    text: string,
   458→    severity: Severity,
   459→    category: string,
   460→    photo: boolean
   461→  ) => {
   462→    const p = jitter({ lat: h.lat, lng: h.lng }, 70, () => rng.float(0, 1));
   463→    const input: ReportCreateInput = {
   464→      description: text,
   465→      category: category as ReportCreateInput["category"],
   466→      severityReported: severity,
   467→      lat: p.lat,
   468→      lng: p.lng,
   469→      addressText: h.name,
   470→      channel: rng.weighted([["APP", 2], ["WEB", 2], ["HOTLINE", 1]] as [string, number][]) as ReportCreateInput["channel"],
   471→      consentGiven: true,
   472→    };
   473→    if (photo) {
   474→      input.photoDataUrl = syntheticPhoto({
   475→        kind: "citizen",
   476→        label: h.name,
   477→        timeLabel: new Date(at).toLocaleString("en-IN", { timeZone: "Asia/Kolkata", day: "numeric", month: "short", hour: "2-digit", minute: "2-digit" }),
   478→      });
   479→      input.photoCaption = "Citizen photo (synthetic)";
   480→    }
   481→    items.push({ at, input });
   482→  };
   483→
   484→  // complaint-light hotspots: structurally vulnerable but under-reported this
   485→  // window (the core research case where frequency-only methods fail)
   486→  const LIGHT = new Set(["HS-MT-02", "HS-ITO-02"]);
   487→
   488→  for (const h of HOTSPOTS) {
   489→    const texts = DESCRIPTIONS[h.profile];
   490→    const cat = categoryForProfile(h.profile);
   491→    if (LIGHT.has(h.code)) {
   492→      // complaint-light structurally-vulnerable spot: 1-2 reports only;
   493→      // risk must come from infrastructure + history, not complaint volume
   494→      if (h.code === "HS-ITO-02") {
   495→        push(new Date(Date.now() - rng.float(30, 33) * HOUR_), h, texts[1], "CRITICAL", cat, false);
   496→        push(new Date(Date.now() - rng.float(8, 14) * HOUR_), h, texts[3], "HIGH", cat, false);
   497→      } else {
   498→        push(new Date(Date.now() - rng.float(30, 34) * HOUR_), h, texts[rng.int(0, texts.length - 1)], "MEDIUM", cat, false);
   499→      }
   500→      continue;
   501→    }
   502→    // wave 1: pre-storm (T-70h)
   503→    push(new Date(Date.now() - (70 + rng.float(-2, 2)) * HOUR_), h, texts[rng.int(0, texts.length - 1)], "MEDIUM", cat, rng.bool(0.2));
   504→    // wave 2: storm peak (T-38..T-28h)
   505→    const wave2 = h.profile === "UNDERPASS" ? 5 : 4;
   506→    for (let i = 0; i < wave2; i++) {
   507→      const at = new Date(Date.now() - rng.float(28, 38) * HOUR_);
   508→      const sev: Severity = rng.weighted([["HIGH", 3], ["CRITICAL", 2]] as [Severity, number][]);
   509→      push(at, h, texts[rng.int(0, texts.length - 1)], sev, cat, rng.bool(0.45));
   510→    }
   511→    // wave 3: aftermath (T-22..T-10h)
   512→    push(new Date(Date.now() - rng.float(10, 22) * HOUR_), h, texts[rng.int(0, texts.length - 1)], rng.bool(0.5) ? "HIGH" : "MEDIUM", cat, rng.bool(0.3));
   513→  }
   514→
   515→  // conflicting-evidence pair at Alipur Road (HS-MT-02): CRITICAL vs LOW
   516→  const mt2 = HOTSPOTS[1];
   517→  push(new Date(Date.now() - 33 * HOUR_), mt2, "Knee-deep water on the low stretch, autos stuck, water entering shops.", "CRITICAL", "WATERLOGGING", true);
   518→  push(new Date(Date.now() - 30 * HOUR_), mt2, "Small puddle on the side of the road, mostly dried now, not a big issue I think.", "LOW", "WATERLOGGING", false);

   775→function nearestHotspot(p: { lat: number; lng: number }): HotspotDef | null {
   776→  let best: HotspotDef | null = null;
   777→  let bestD = Infinity;
   778→  for (const h of HOTSPOTS) {
   779→    const d = distanceM(p, { lat: h.lat, lng: h.lng });
   780→    if (d < bestD) {
   781→      bestD = d;
   782→      best = h;
   783→    }
   784→  }
   785→  return best;
   786→}
