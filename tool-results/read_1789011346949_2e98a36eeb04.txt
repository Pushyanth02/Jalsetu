   105→   105→export function InvestigateView() {
   106→   106→  const [tool, setTool] = useState<ToolId>("classify");
   107→   107→  const { toast } = useToast();
   108→   108→  const qc = useQueryClient();
   109→   109→
   110→   110→  const healthQ = useQuery({
   111→   111→    queryKey: ["ai-health"],
   112→   112→    queryFn: () => apiGet<{ ai: ProviderHealthInfo }>("/api/health").then((r) => r.data.ai),
   113→   113→    refetchInterval: 120_000,
   114→   114→    staleTime: 60_000,
   115→   115→  });
   116→   116→
   117→   117→  // freshest health wins: compare the run-time probe vs the polled snapshot
   118→   118→  const [toolHealth, setToolHealth] = useState<ProviderHealthInfo | null>(null);
   119→   119→  const health = useMemo(() => {
   120→   120→    if (toolHealth && healthQ.data) {
   121→   121→      const tRun = toolHealth.lastCheckedAt ? Date.parse(toolHealth.lastCheckedAt) : 0;
   122→   122→      const tPolled = healthQ.data.lastCheckedAt ? Date.parse(healthQ.data.lastCheckedAt) : 0;
   123→   123→      return tRun >= tPolled ? toolHealth : healthQ.data;
   124→   124→    }
   125→   125→    return toolHealth ?? healthQ.data ?? null;
   126→   126→  }, [toolHealth, healthQ.data]);
   127→   127→
   128→   128→  // --- Tool A: classification (stored report or ad-hoc text) ---
   129→   129→  const classify = useMutation({
   130→   130→    mutationFn: (vars: ClassifyVars) =>
   131→   131→      apiPost<ClassifyResponse>("/api/ai/classify", {
   132→   132→        reportId: vars.reportId,
   133→   133→        description: vars.description,
   134→   134→        severityReported: vars.severityReported,
   135→   135→        lat: vars.lat,
   136→   136→        lng: vars.lng,
   137→   137→      }, "ANALYST").then((r) => ({ ...r.data, _label: vars.reportRef ?? "ad-hoc text" })),
   138→   138→    onSuccess: (d) => {
   139→   139→      setToolHealth(d.provenance.providerHealth);
   140→   140→      qc.invalidateQueries({ queryKey: ["reports"] });
   141→   141→      toast({
   142→   142→        title: "Classification complete",
   143→   143→        description: `${d.classification.category.toLowerCase()} · severity ${d.classification.severity.toLowerCase()} · provider ${d.provenance.provider}${d.provenance.fallbackUsed ? " (fallback)" : ""}.`,
   144→   144→      });
   145→   145→    },
   146→   146→    onError: (e) => toast({ title: "Classification failed", description: (e as ApiClientError).message, variant: "destructive" }),
   147→   147→  });
   148→   148→
   149→   149→  // --- Tool B: duplicate clustering ---
   150→   150→  const [radius, setRadius] = useState<number>(150);
   151→   151→  const cluster = useMutation({
   152→   152→    mutationFn: (rm: number) =>
   153→   153→      apiPost<ClusterResponse>("/api/ai/cluster", { radiusM: rm }, "ANALYST").then((r) => ({ data: r.data, meta: r.meta ?? {}, radius: rm })),
   154→   154→    onSuccess: (d) => {
   155→   155→      toast({
   156→   156→        title: "Clustering complete",
   157→   157→        description: `${d.data.clusters.length} clusters, ${d.data.singletons} singletons across ${String(d.meta.evaluated ?? "?")} reports (last 72h).`,
   158→   158→      });
   159→   159→    },
   160→   160→    onError: (e) => toast({ title: "Clustering failed", description: (e as ApiClientError).message, variant: "destructive" }),
   161→   161→  });
   162→   162→
   163→   163→  // --- Tool C: risk assessment ---
   164→   164→  const risk = useMutation({
   165→   165→    mutationFn: (eventId: string) => apiPost<RiskResponse>("/api/ai/risk", { eventId }, "ANALYST").then((r) => r.data),
   166→   166→    onSuccess: (d) => {
   167→   167→      setToolHealth(d.provenance.advisoryProviderHealth);
   168→   168→      qc.invalidateQueries({ queryKey: ["events"] });
   169→   169→      qc.invalidateQueries({ queryKey: ["event"] });
   170→   170→      qc.invalidateQueries({ queryKey: ["overview"] });
   171→   171→      toast({
   172→   172→        title: "Risk assessed",
   173→   173→        description: `Score ${d.risk.score} (${d.risk.band.toLowerCase()}). Advisory provider: ${d.advisory.provider}${d.advisory.fallbackUsed ? " (fallback)" : ""}.`,
   174→   174→      });
   175→   175→    },
   176→   176→    onError: (e) => toast({ title: "Risk assessment failed", description: (e as ApiClientError).message, variant: "destructive" }),
   177→   177→  });
   178→   178→
   179→   179→  return (
   180→   180→    <div className="flex flex-1 min-h-0 flex-col">
   181→   181→      {/* provider health summary strip */}
   182→   182→      <div className="hairline-b bg-ink-900/30 px-4 sm:px-5 py-2.5">
   183→   183→        <div className="flex flex-wrap items-center gap-x-4 gap-y-2">
   184→   184→          <span className="flex items-center gap-2 shrink-0">
   185→   185→            <Radio className="size-3.5 text-water" aria-hidden />
   186→   186→            <span className="micro-label !text-[0.62rem] text-foreground/80">ai investigation</span>
   187→   187→          </span>
   188→   188→          {health ? (
   189→   189→            <>
   190→   190→              <ProviderChip provider={health.provider} model={health.modelId} />
   191→   191→              <span className={cn("micro-label !text-[0.52rem] flex items-center gap-1.5", health.available ? "text-verified" : "text-sev-high")}>
   192→   192→                <span aria-hidden className={cn("size-1.5 rounded-full", health.available ? "bg-verified" : "bg-sev-high")} />
   193→   193→                {health.available ? "available" : "unavailable"}
   194→   194→              </span>
   195→   195→              <span className="micro-label !text-[0.52rem] text-muted-foreground/70 flex items-center gap-1.5">
   196→   196→                configured by {health.configuredBy.toLowerCase()}
   197→   197→                {health.lastCheckedAt && <>· probed <TimeAgo iso={health.lastCheckedAt} /></>}
   198→   198→              </span>
   199→   199→              {health.lastError && (
   200→   200→                <span className="text-[0.62rem] text-sev-moderate" role="note" title={health.lastError}>
   201→   201→                  last error: {health.lastError}
   202→   202→                </span>
   203→   203→              )}
   204→   204→            </>
   205→   205→          ) : healthQ.isLoading ? (
   206→   206→            <span className="micro-label !text-[0.52rem] text-muted-foreground/60">probing provider…</span>
   207→   207→          ) : (
   208→   208→            <span className="micro-label !text-[0.52rem] text-sev-critical">provider health unavailable</span>
   209→   209→          )}
   210→   210→          <span className="micro-label ml-auto !text-[0.5rem] text-muted-foreground/60 hidden sm:inline">
   211→   211→            structured output only · no chain-of-thought exposed
   212→   212→          </span>

   216→   216→      {/* tool grid: selector + inputs (left) | results (right) */}

   266→   266→  const [description, setDescription] = useState("");

   273→   273→  const latNum = Number(lat);
   274→   274→  const lngNum = Number(lng);
   275→   275→  const coordsOk = (lat.trim() === "" || Number.isFinite(latNum)) && (lng.trim() === "" || Number.isFinite(lngNum));

   311→   311→            />

   313→   313→          </div>
   314→   314→          <div className="grid grid-cols-3 gap-2">
   315→   315→            <div className="space-y-1">
   316→   316→              <Label className="text-[0.65rem]">Severity</Label>

   323→   323→            </div>
   324→   324→            <div className="space-y-1">

   335→   335→            disabled={pending || !descOk || !coordsOk}
   336→   336→            onClick={() =>
   337→   337→              classify.mutate({
   338→   338→                description: description.trim(),
   339→   339→                severityReported: severity,
   340→   340→                lat: lat.trim() && Number.isFinite(latNum) ? latNum : undefined,

   344→   344→            className="w-full bg-water text-ink-950 hover:bg-water/85"

   352→   352→        </div>
   353→   353→      )}
   354→   354→
   355→   355→      {classify.isError && (
   356→   356→        <p className="micro-label !text-[0.52rem] text-sev-critical/90" role="alert">
   357→   357→          last run failed: {classify.error.message}
   358→   358→        </p>
   359→   359→      )}
   360→   360→    </div>
   361→   361→  );
   362→   362→}

   364→   364→function ReportQueue({ pending, onClassify }: { pending: boolean; onClassify: (r: ReportRow) => void }) {
   365→   365→  const reportsQ = useQuery({
   366→   366→    queryKey: ["reports", 50],
   367→   367→    queryFn: () => apiGet<ReportRow[]>("/api/reports?limit=50").then((r) => r.data),
   368→   368→    refetchInterval: 60_000,
   369→   369→    staleTime: 30_000,
   370→   370→  });
   371→   371→
   372→   372→  if (reportsQ.isLoading) return <LoadingRows rows={6} />;
   373→   373→  if (reportsQ.isError) return <ErrorNote message={(reportsQ.error as Error).message} onRetry={() => reportsQ.refetch()} />;
   374→   374→
   375→   375→  const reports = reportsQ.data ?? [];
   376→   376→  if (reports.length === 0)
   377→   377→    return <EmptyState title="No reports in the queue" hint="Citizen reports appear here as they arrive." />;
   378→   378→
   379→   379→  return (
   380→   380→    <div>
   381→   381→      <p className="micro-label !text-[0.52rem] text-muted-foreground/70 mb-2">
   382→   382→        recent reports · newest first · {reports.length}
   383→   383→      </p>
   384→   384→      <ul>
