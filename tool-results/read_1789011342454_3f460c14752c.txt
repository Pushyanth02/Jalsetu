    20→// AI INVESTIGATION - three structured assistance tools (classification,
    21→// duplicate clustering, risk assessment) with honest provenance and honest
    22→// fallback reporting. Structured output only; no chain-of-thought is ever
    23→// requested, returned, or displayed.
    24→
    25→// --- API types -----------------------------------------------------------------------
    26→
    27→interface ReportRow {
    28→  id: string; publicRef: string; description: string; category: string; severityReported: string;
    29→  lat: number; lng: number; addressText: string | null; submittedAt: string; channel: string;
    30→  status: string; isDuplicate: boolean; urbanEventId: string | null; source: string;
    31→  classificationProvider: string | null; classificationModel: string | null; classificationConfidence: number | null;
    32→  classification: { provider: string; modelId: string | null; confidence: number | null } | null;
    33→}
    34→
    35→interface ProviderHealthInfo {
    36→  provider: string; modelId: string; available: boolean;
    37→  lastCheckedAt: string | null; lastError: string | null; configuredBy: string;
    38→}
    39→
    40→interface ClassifyResponse {
    41→  classification: {
    42→    category: string; severity: string; confidence: number;
    43→    isDuplicateSuspected: boolean; duplicateReason?: string;
    44→    factors: string[]; summary: string;
    45→  };
    46→  provenance: {
    47→    provider: string; modelId: string; latencyMs: number;
    48→    fallbackUsed: boolean; fallbackReason: string | null;
    49→    providerHealth: ProviderHealthInfo; version: string;
    50→  };
    51→  context: {
    52→    rainfall24hMm: number | null; rainfall72hMm: number | null;
    53→    nearestStation: { code: string; name: string; distanceM: number } | null;
    54→    historicalIncidentsNearby: number;
    55→    nearestAssets: { code: string; kind: string; distanceM: number; condition: number | null }[];
    56→    nearbyOpenEvents: number;
    57→  };
    58→  duplicateSignals: { code: string; distanceM: number; hoursApart: number; recurrence: boolean; reason: string }[];
    59→}
    60→
    61→interface ClusterResponse {
    62→  clusters: {
    63→    id: string; centroid: { lat: number; lng: number }; size: number; spatialDiameterM: number;
    64→    members: {
    65→      id: string; lat: number; lng: number; publicRef: string; category: string;
    66→      severityReported: string; description: string; submittedAt: string; urbanEventId: string | null;
    67→    }[];
    68→  }[];
    69→  singletons: number;
    70→}
    71→
    72→interface RiskResponse {
    73→  event: { id: string; code: string; title: string };
    74→  risk: {
    75→    score: number; band: string; modelVersion: string; computedAt: string;
    76→    factors: { key: string; label: string; value: number; weight: number; contribution: number; raw: string }[];
    77→  };
    78→  advisory: {
    79→    output: {
    80→      narrative: string;
    81→      factors?: { key: string; label: string; direction: string; weight: number }[];
    82→      recommendedInvestigation?: string;
    83→    };
    84→    provider: string; modelId: string; fallbackUsed: boolean; error: string | null;
    85→  };
    86→  provenance: { riskModelVersion: string; advisoryProviderHealth: ProviderHealthInfo };
    87→}
    88→
    89→type ToolId = "classify" | "cluster" | "risk";
    90→
    91→type ClassifyVars = { reportId?: string; reportRef?: string; description?: string; severityReported?: string; lat?: number; lng?: number };
    92→type ClassifyMutation = UseMutationResult<ClassifyResponse & { _label: string }, Error, ClassifyVars>;
    93→type ClusterMutation = UseMutationResult<{ data: ClusterResponse; meta: Record<string, unknown>; radius: number }, Error, number>;
    94→type RiskMutation = UseMutationResult<RiskResponse, Error, string>;

   105→export function InvestigateView() {
   106→  const [tool, setTool] = useState<ToolId>("classify");
   107→  const { toast } = useToast();
   108→  const qc = useQueryClient();
   109→
   110→  const healthQ = useQuery({
   111→    queryKey: ["ai-health"],
   112→    queryFn: () => apiGet<{ ai: ProviderHealthInfo }>("/api/health").then((r) => r.data.ai),
   113→    refetchInterval: 120_000,
   114→    staleTime: 60_000,
   115→  });
   116→
   117→  // freshest health wins: compare the run-time probe vs the polled snapshot
   118→  const [toolHealth, setToolHealth] = useState<ProviderHealthInfo | null>(null);
   119→  const health = useMemo(() => {
   120→    if (toolHealth && healthQ.data) {
   121→      const tRun = toolHealth.lastCheckedAt ? Date.parse(toolHealth.lastCheckedAt) : 0;
   122→      const tPolled = healthQ.data.lastCheckedAt ? Date.parse(healthQ.data.lastCheckedAt) : 0;
   123→      return tRun >= tPolled ? toolHealth : healthQ.data;
   124→    }
   125→    return toolHealth ?? healthQ.data ?? null;
   126→  }, [toolHealth, healthQ.data]);
   127→
   128→  // --- Tool A: classification (stored report or ad-hoc text) ---
   129→  const classify = useMutation({
   130→    mutationFn: (vars: ClassifyVars) =>
   131→      apiPost<ClassifyResponse>("/api/ai/classify", {
   132→        reportId: vars.reportId,
   133→        description: vars.description,
   134→        severityReported: vars.severityReported,
   135→        lat: vars.lat,
   136→        lng: vars.lng,
   137→      }, "ANALYST").then((r) => ({ ...r.data, _label: vars.reportRef ?? "ad-hoc text" })),
   138→    onSuccess: (d) => {
   139→      setToolHealth(d.provenance.providerHealth);
   140→      qc.invalidateQueries({ queryKey: ["reports"] });
   141→      toast({
   142→        title: "Classification complete",
   143→        description: `${d.classification.category.toLowerCase()} · severity ${d.classification.severity.toLowerCase()} · provider ${d.provenance.provider}${d.provenance.fallbackUsed ? " (fallback)" : ""}.`,
   144→      });
   145→    },
   146→    onError: (e) => toast({ title: "Classification failed", description: (e as ApiClientError).message, variant: "destructive" }),
   147→  });
   148→
   149→  // --- Tool B: duplicate clustering ---
   150→  const [radius, setRadius] = useState<number>(150);
   151→  const cluster = useMutation({
   152→    mutationFn: (rm: number) =>
   153→      apiPost<ClusterResponse>("/api/ai/cluster", { radiusM: rm }, "ANALYST").then((r) => ({ data: r.data, meta: r.meta ?? {}, radius: rm })),
   154→    onSuccess: (d) => {
   155→      toast({
   156→        title: "Clustering complete",
   157→        description: `${d.data.clusters.length} clusters, ${d.data.singletons} singletons across ${String(d.meta.evaluated ?? "?")} reports (last 72h).`,
   158→      });
   159→    },
   160→    onError: (e) => toast({ title: "Clustering failed", description: (e as ApiClientError).message, variant: "destructive" }),
   161→  });
   162→
   163→  // --- Tool C: risk assessment ---
   164→  const risk = useMutation({
   165→    mutationFn: (eventId: string) => apiPost<RiskResponse>("/api/ai/risk", { eventId }, "ANALYST").then((r) => r.data),
   166→    onSuccess: (d) => {
   167→      setToolHealth(d.provenance.advisoryProviderHealth);
   168→      qc.invalidateQueries({ queryKey: ["events"] });
   169→      qc.invalidateQueries({ queryKey: ["event"] });
   170→      qc.invalidateQueries({ queryKey: ["overview"] });
   171→      toast({
   172→        title: "Risk assessed",
   173→        description: `Score ${d.risk.score} (${d.risk.band.toLowerCase()}). Advisory provider: ${d.advisory.provider}${d.advisory.fallbackUsed ? " (fallback)" : ""}.`,
   174→      });
   175→    },
   176→    onError: (e) => toast({ title: "Risk assessment failed", description: (e as ApiClientError).message, variant: "destructive" }),
   177→  });
   178→
   179→  return (
   180→    <div className="flex flex-1 min-h-0 flex-col">
   181→      {/* provider health summary strip */}
   182→      <div className="hairline-b bg-ink-900/30 px-4 sm:px-5 py-2.5">
   183→        <div className="flex flex-wrap items-center gap-x-4 gap-y-2">
   184→          <span className="flex items-center gap-2 shrink-0">
   185→            <Radio className="size-3.5 text-water" aria-hidden />
   186→            <span className="micro-label !text-[0.62rem] text-foreground/80">ai investigation</span>
   187→          </span>
   188→          {health ? (
   189→            <>
   190→              <ProviderChip provider={health.provider} model={health.modelId} />
   191→              <span className={cn("micro-label !text-[0.52rem] flex items-center gap-1.5", health.available ? "text-verified" : "text-sev-high")}>
   192→                <span aria-hidden className={cn("size-1.5 rounded-full", health.available ? "bg-verified" : "bg-sev-high")} />
   193→                {health.available ? "available" : "unavailable"}
   194→              </span>
   195→              <span className="micro-label !text-[0.52rem] text-muted-foreground/70 flex items-center gap-1.5">
   196→                configured by {health.configuredBy.toLowerCase()}
   197→                {health.lastCheckedAt && <>· probed <TimeAgo iso={health.lastCheckedAt} /></>}
   198→              </span>
   199→              {health.lastError && (
   200→                <span className="text-[0.62rem] text-sev-moderate" role="note" title={health.lastError}>
   201→                  last error: {health.lastError}
   202→                </span>
   203→              )}
   204→            </>
   205→          ) : healthQ.isLoading ? (
   206→            <span className="micro-label !text-[0.52rem] text-muted-foreground/60">probing provider…</span>
   207→          ) : (
   208→            <span className="micro-label !text-[0.52rem] text-sev-critical">provider health unavailable</span>
   209→          )}
   210→          <span className="micro-label ml-auto !text-[0.5rem] text-muted-foreground/60 hidden sm:inline">
   211→            structured output only · no chain-of-thought exposed
   212→          </span>
   213→        </div>
   214→      </div>
   215→
   216→      {/* tool grid: selector + inputs (left) | results (right) */}
   217→      <div className="flex-1 min-h-0 grid grid-cols-1 lg:grid-cols-[350px_1fr]">
   218→        <aside className="lg:hairline-r bg-ink-900/20 flex flex-col min-h-0 border-b lg:border-b-0">
   219→          {/* tool selector */}
   220→          <div className="grid grid-cols-3 lg:grid-cols-1 hairline-b shrink-0">
   221→            {TOOLS.map((t) => {
   222→              const active = tool === t.id;
   223→              return (
   224→                <button
   225→                  key={t.id}
   226→                  onClick={() => setTool(t.id)}
   227→                  aria-pressed={active}
   228→                  className={cn(
   229→                    "text-left px-3.5 py-2.5 lg:py-3 transition-colors hairline-r last:border-r-0 lg:hairline-b lg:border-r-0",
   230→                    active ? "bg-water/8" : "hover:bg-ink-850/60"
   231→                  )}
   232→                >
   233→                  <span className="flex items-center gap-2">
   234→                    <t.icon className={cn("size-3.5 shrink-0", active ? "text-water" : "text-muted-foreground")} aria-hidden />
   235→                    <span className={cn("micro-label !text-[0.58rem]", active ? "text-water" : "text-foreground/70")}>{t.label}</span>
   236→                  </span>
   237→                  <span className="hidden lg:block mt-0.5 text-[0.62rem] text-muted-foreground/80 leading-snug pl-5.5">{t.hint}</span>
   238→                </button>
   239→              );
   240→            })}
   241→          </div>
   242→
   243→          {/* per-tool input area */}
   244→          <div className="flex-1 min-h-0 overflow-y-auto">
   245→            {tool === "classify" && <ClassifyInput classify={classify} />}
   246→            {tool === "cluster" && <ClusterInput cluster={cluster} radius={radius} setRadius={setRadius} />}
   247→            {tool === "risk" && <RiskInput risk={risk} />}
   248→          </div>
   249→        </aside>
   250→
   251→        {/* results column */}
   252→        <div className="min-h-0 overflow-y-auto p-3 sm:p-4 space-y-3">
   253→          {tool === "classify" && <ClassifyResults m={classify} />}
   254→          {tool === "cluster" && <ClusterResults m={cluster} />}
   255→          {tool === "risk" && <RiskResults m={risk} />}
   256→        </div>
   257→      </div>
   258→    </div>
   259→  );
   260→}

   264→function ClassifyInput({ classify }: { classify: ClassifyMutation }) {
   265→  const [mode, setMode] = useState<"queue" | "adhoc">("queue");
   266→  const [description, setDescription] = useState("");
   267→  const [severity, setSeverity] = useState<string>("MEDIUM");
   268→  const [lat, setLat] = useState("");
   269→  const [lng, setLng] = useState("");
   270→
   271→  const pending = classify.isPending;
   272→  const descOk = description.trim().length >= 12;
   273→  const latNum = Number(lat);
   274→  const lngNum = Number(lng);
   275→  const coordsOk = (lat.trim() === "" || Number.isFinite(latNum)) && (lng.trim() === "" || Number.isFinite(lngNum));
   276→
   277→  return (
   278→    <div className="p-3.5 space-y-3">
   279→      {/* mode toggle */}
   280→      <div className="flex rounded-sm border border-border/70 overflow-hidden" role="tablist" aria-label="Classification input mode">
   281→        {(["queue", "adhoc"] as const).map((m) => (
   282→          <button
   283→            key={m}
   284→            role="tab"
   285→            aria-selected={mode === m}
   286→            onClick={() => setMode(m)}
   287→            className={cn(
   288→              "flex-1 micro-label !text-[0.55rem] py-1.5 transition-colors",
   289→              mode === m ? "bg-water/12 text-water" : "text-muted-foreground hover:text-foreground"
   290→            )}
   291→          >
   292→            {m === "queue" ? "report queue" : "ad-hoc text"}
   293→          </button>
   294→        ))}
   295→      </div>
   296→
   297→      {mode === "queue" ? (
   298→        <ReportQueue pending={pending} onClassify={(r) => classify.mutate({ reportId: r.id, reportRef: r.publicRef })} />
   299→      ) : (
   300→        <div className="space-y-3">
   301→          <div className="space-y-1.5">
   302→            <Label htmlFor="adhoc-desc" className="text-xs">Description</Label>
   303→            <Textarea
   304→              id="adhoc-desc"
   305→              rows={4}
   306→              value={description}
   307→              onChange={(e) => setDescription(e.target.value)}
   308→              placeholder="e.g. knee-deep water under the railway bridge, autos stalled, drain overflowing"
   309→              className="bg-ink-900 border-border text-xs leading-relaxed"
   310→              maxLength={600}
   311→            />
   312→            <p className="text-[0.6rem] text-muted-foreground/70">min 12 characters · no personal identifiers</p>
   313→          </div>
   314→          <div className="grid grid-cols-3 gap-2">
   315→            <div className="space-y-1">
   316→              <Label className="text-[0.65rem]">Severity</Label>
   317→              <Select value={severity} onValueChange={setSeverity}>
   318→                <SelectTrigger className="h-8 bg-ink-900 border-border text-xs" aria-label="Reported severity"><SelectValue /></SelectTrigger>
   319→                <SelectContent>
   320→                  {SEVERITIES.map((s) => <SelectItem key={s} value={s} className="text-xs">{s.toLowerCase()}</SelectItem>)}
   321→                </SelectContent>
   322→              </Select>
   323→            </div>
   324→            <div className="space-y-1">
   325→              <Label htmlFor="adhoc-lat" className="text-[0.65rem]">Lat (opt)</Label>
   326→              <Input id="adhoc-lat" inputMode="decimal" value={lat} onChange={(e) => setLat(e.target.value)} placeholder="28.6129" className="h-8 bg-ink-900 border-border text-xs font-mono" />
   327→            </div>
   328→            <div className="space-y-1">
   329→              <Label htmlFor="adhoc-lng" className="text-[0.65rem]">Lng (opt)</Label>
   330→              <Input id="adhoc-lng" inputMode="decimal" value={lng} onChange={(e) => setLng(e.target.value)} placeholder="77.2295" className="h-8 bg-ink-900 border-border text-xs font-mono" />
   331→            </div>
   332→          </div>
   333→          <Button
   334→            size="sm"
   335→            disabled={pending || !descOk || !coordsOk}
   336→            onClick={() =>
   337→              classify.mutate({
   338→                description: description.trim(),
   339→                severityReported: severity,
   340→                lat: lat.trim() && Number.isFinite(latNum) ? latNum : undefined,
   341→                lng: lng.trim() && Number.isFinite(lngNum) ? lngNum : undefined,
   342→              })
   343→            }
   344→            className="w-full bg-water text-ink-950 hover:bg-water/85"
   345→          >
   346→            <Sparkles className={cn("size-3.5", pending && "animate-pulse")} aria-hidden />
   347→            {pending ? "Classifying…" : "Classify text"}
   348→          </Button>
   349→          <p className="text-[0.6rem] text-muted-foreground/70 leading-relaxed">
   350→            Delhi bounds: lat 28.3 to 28.9, lng 76.8 to 77.6. Coordinates outside Delhi are rejected by validation.
   351→          </p>
   352→        </div>
   353→      )}
   354→
   355→      {classify.isError && (
   356→        <p className="micro-label !text-[0.52rem] text-sev-critical/90" role="alert">
   357→          last run failed: {classify.error.message}
   358→        </p>
   359→      )}
   360→    </div>
   361→  );
   362→}

   364→function ReportQueue({ pending, onClassify }: { pending: boolean; onClassify: (r: ReportRow) => void }) {
   365→  const reportsQ = useQuery({
   366→    queryKey: ["reports", 50],
   367→    queryFn: () => apiGet<ReportRow[]>("/api/reports?limit=50").then((r) => r.data),
   368→    refetchInterval: 60_000,
   369→    staleTime: 30_000,
   370→  });
   371→
   372→  if (reportsQ.isLoading) return <LoadingRows rows={6} />;
   373→  if (reportsQ.isError) return <ErrorNote message={(reportsQ.error as Error).message} onRetry={() => reportsQ.refetch()} />;
   374→
   375→  const reports = reportsQ.data ?? [];
   376→  if (reports.length === 0)
   377→    return <EmptyState title="No reports in the queue" hint="Citizen reports appear here as they arrive." />;
   378→
   379→  return (
   380→    <div>
   381→      <p className="micro-label !text-[0.52rem] text-muted-foreground/70 mb-2">
   382→        recent reports · newest first · {reports.length}
   383→      </p>
   384→      <ul>
   385→        {reports.map((r) => (
   386→          <li key={r.id} className="hairline-b last:border-0 py-2.5 first:pt-0">
   387→            <div className="flex items-center gap-2 flex-wrap">
   388→              <span className="data-mono text-[0.65rem] font-semibold text-water/90">{r.publicRef}</span>
   389→              <span className="micro-label !text-[0.5rem]">{r.channel.toLowerCase()}</span>
   390→              <span className={cn("micro-label !text-[0.5rem]", sevTone(r.severityReported))}>{r.severityReported.toLowerCase()}</span>
   391→              {r.isDuplicate && <span className="micro-label !text-[0.5rem] text-sev-moderate/80">merged</span>}
   392→              <TimeAgo iso={r.submittedAt} />
   393→            </div>
   394→            <p className="mt-0.5 text-[0.72rem] text-foreground/85 leading-snug line-clamp-2">{r.description}</p>
   395→            <div className="mt-1.5 flex items-center gap-2 flex-wrap">
   396→              <SourceBadge source={r.source} />
   397→              {r.classification && <ProviderChip provider={r.classification.provider} model={r.classification.modelId} />}
   398→              <Button
   399→                size="sm"
   400→                variant="outline"
   401→                disabled={pending}
   402→                onClick={() => onClassify(r)}
   403→                className="ml-auto h-7 text-[0.65rem] border-water/40 text-water hover:bg-water/10 hover:text-water"
   404→              >
   405→                <Sparkles className="size-3" aria-hidden />
   406→                {r.classification ? "Re-run" : "Classify"}
   407→              </Button>
   408→            </div>
   409→          </li>
   410→        ))}
   411→      </ul>
   412→    </div>
   413→  );
   414→}

   416→function ClassifyResults({ m }: { m: ClassifyMutation }) {
   417→  if (m.isPending) {
   418→    return (
   419→      <Panel title="classification · result">
   420→        <LoadingRows rows={5} />
   421→        <p className="micro-label !text-[0.52rem] text-muted-foreground/60 mt-2">
   422→          running provider call for {m.variables?.reportRef ?? "ad-hoc text"}…
   423→        </p>
   424→      </Panel>
   425→    );
   426→  }
   427→  if (m.isError) {
   428→    return (
   429→      <Panel title="classification · result">
   430→        <ErrorNote
   431→          message={(m.error as Error).message}
   432→          onRetry={m.variables ? () => m.mutate(m.variables!) : undefined}
   433→        />
   434→      </Panel>
   435→    );
   436→  }
   437→  if (!m.data) {
   438→    return (
   439→      <Panel title="classification · result">
   440→        <EmptyState
   441→          icon={<Sparkles className="size-7" />}
   442→          title="No classification run yet"
   443→          hint="Pick a report from the queue, or switch to ad-hoc text mode, then run the classifier."
   444→        />
   445→      </Panel>
   446→    );
   447→  }
   448→
   449→  const r = m.data;
   450→  const cl = r.classification;
   451→  const prov = r.provenance;
   452→
   453→  return (
   454→    <>
   455→      <Panel
   456→        title={
   457→          <span className="flex items-center gap-2">
   458→            <Sparkles className="size-3 text-water" aria-hidden />
   459→            classification · {r._label}
   460→          </span>
   461→        }

   897→                  f.direction === "AGGRAVATES" ? "border-sev-high/30 text-sev-high/90 bg-sev-high/6" : "border-verified/30 text-verified/90 bg-verified/6"
   898→                )}
   899→              >
   900→                {f.direction === "AGGRAVATES" ? "▲" : "▼"} {f.label} · w {f.weight.toFixed(2)}
   901→              </li>
   902→            ))}
   903→          </ul>
   904→        )}
   905→        {r.advisory.output.recommendedInvestigation && (
   906→          <p className="mt-3 text-xs text-water/90">
   907→            <span className="micro-label !text-[0.55rem] mr-1.5">recommended</span>{" "}
   908→            {r.advisory.output.recommendedInvestigation}
   909→          </p>
   910→        )}
   911→        <p className="mt-3 micro-label !text-[0.5rem] text-muted-foreground/60">
   912→          Structured output only. No chain-of-thought exposed.
   913→        </p>
   914→      </Panel>
   915→
   916→      <Panel title="provenance">
   917→        <dl className="space-y-2 text-xs">
   918→          <FactRow label="risk engine"><span className="data-mono text-[0.62rem]">{r.provenance.riskModelVersion}</span></FactRow>
   919→          <FactRow label="advisory provider"><ProviderChip provider={r.advisory.provider} model={r.advisory.modelId} /></FactRow>
   920→          <FactRow label="fallback">
   921→            <span className={cn("micro-label !text-[0.55rem]", r.advisory.fallbackUsed ? "text-sev-moderate" : "text-verified")}>
   922→              {r.advisory.fallbackUsed ? "used" : "not used"}
   923→            </span>
   924→          </FactRow>
   925→          <FactRow label="event">
   926→            <button onClick={() => navigate("event", r.event.code)} className="data-mono text-[0.62rem] text-water hover:text-foreground transition-colors">
   927→              {r.event.code} →
   928→            </button>
   929→          </FactRow>
   930→        </dl>
   931→      </Panel>
   932→    </>
   933→  );
   934→}
