   724→function whatThisMeans(b: AnalyticsData, p: ProposedData): string[] {
   725→  const f1Delta = p.metrics.f1 - b.metrics.f1;
   726→  const bestProposedK = p.byK.reduce((best, r) => (r.f1 > best.f1 ? r : best), p.byK[0]);
   727→  const k12 = b.byK.find((r) => r.k === 12);
   728→  const p12 = p.byK.find((r) => r.k === 12);
   729→  const parts: string[] = [];
   730→
   731→  parts.push(
   732→    `On this seeded pilot (${p.dataset.reports} reports, ${p.dataset.events} events, ${p.dataset.groundTruthHotspots} known hotspots across ${p.dataset.candidateCells.toLocaleString("en-IN")} candidate cells), the evidence-integrated ranking reached F1 ${p.metrics.f1.toFixed(2)} versus ${b.metrics.f1.toFixed(2)} for complaint frequency alone, and matched ${Math.round(p.metrics.spatialHitRate * 100)}% of ground-truth sites within 250m versus ${Math.round(b.metrics.spatialHitRate * 100)}% for the baseline. Both approaches were scored on identical candidates and labels, so the gap is attributable to the scoring function, not the data.`
   733→  );
   734→
   735→  parts.push(
   736→    `The AUC margin is small (${p.metrics.auc.toFixed(3)} vs ${b.metrics.auc.toFixed(3)}) because both rankers separate the ${p.dataset.positives} positive cells from the bulk; the improvement concentrates at the operating point. By K, the proposed ranking is strongest at k=${bestProposedK.k} (F1 ${bestProposedK.f1.toFixed(2)})${k12 && p12 ? `, and at k=12 both approaches return F1 ${k12.f1.toFixed(2)} as prediction lists thin out under 400m non-maximum suppression` : ""}.`
   737→  );
   738→
   739→  if (f1Delta >= 0.15) {
   740→    parts.push(
   741→      `These are synthetic results computed against seeded ground truth with known labels. They show that the evaluation harness can discriminate the two scoring functions on pilot data, not that the proposed approach would hold the same margins (${f1Delta.toFixed(2)} F1 here) on real city data.`
   742→    );
   743→  } else {
   744→    parts.push(
   745→      `These are synthetic results computed against seeded ground truth with known labels and should not be read as real-world effect sizes.`
   746→    );
   747→  }
   748→
   749→  return parts;
   750→}

   776→function matchHotspots(data?: HotspotResponse): {
   777→  computed: ComputedWithMatch[];
   778→  groundTruth: GroundTruthWithMatch[];
   779→  covered: number;
   780→} {
   781→  if (!data) return { computed: [], groundTruth: [], covered: 0 };
   782→  const computed: ComputedWithMatch[] = data.computed.map((c) => {
   783→    let nearestD: number | null = null;
   784→    let nearest: { code: string; name: string } | undefined;
   785→    for (const g of data.groundTruth) {
   786→      const d = distanceM({ lat: c.lat, lng: c.lng }, { lat: g.lat, lng: g.lng });
   787→      if (nearestD == null || d < nearestD) {
   788→        nearestD = d;
   789→        nearest = { code: g.code, name: g.name };
   790→      }
   791→    }
   792→    return { ...c, nearestD, nearest };
   793→  });
   794→  const groundTruth: GroundTruthWithMatch[] = data.groundTruth.map((g) => {
   795→    let nearestD: number | null = null;
   796→    for (const c of data.computed) {
   797→      const d = distanceM({ lat: c.lat, lng: c.lng }, { lat: g.lat, lng: g.lng });
   798→      if (nearestD == null || d < nearestD) nearestD = d;
   799→    }
   800→    return { ...g, nearestD };
   801→  });
   802→  const covered = groundTruth.filter((g) => g.nearestD != null && g.nearestD <= MATCH_RADIUS_M).length;
   803→  return { computed, groundTruth, covered };
   804→}
