   119→interface PingState {

   121→  status: "pending" | "ok" | "err";
   122→  ms: number | null;

   125→
   126→async function ping(path: string): Promise<PingState> {
   127→  const t0 = performance.now();
   128→  try {
   129→    const res = await fetch(path, { cache: "no-store" });
   130→    const ms = Math.round(performance.now() - t0);
   131→    return { path, status: res.ok ? "ok" : "err", ms, code: res.status };
   132→  } catch {
   133→    return { path, status: "err", ms: Math.round(performance.now() - t0), code: null };
   134→  }
   135→}

   749→function SourceStatusBadge({ status }: { status: string }) {
   750→  const meta =
   751→    status === "OK"

   753→      : status === "STALE"

   755→        : { label: "EMPTY", cls: "text-red-700 border-red-200 bg-red-50", dot: "bg-red-500" };

   757→    <span className={cn("inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 micro-label !text-[0.6rem] !tracking-[0.1em] font-medium", meta.cls)}>

   764→function RunStatusBadge({ status }: { status: string }) {
   765→  const meta =
   766→    status === "SUCCEEDED"
   767→      ? { label: "succeeded", cls: "text-emerald-700 border-emerald-200 bg-emerald-50", dot: "bg-emerald-500" }

   773→  return (
   774→    <span className={cn("inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 micro-label !text-[0.6rem] !tracking-[0.08em] font-medium", meta.cls)}>
