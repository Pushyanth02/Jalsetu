    74→export function EventDetailView() {
    75→  const eventId = useUi((s) => s.eventId);
    76→  const { toast } = useToast();
    77→  const qc = useQueryClient();
    78→
    79→  const detailQ = useQuery({
    80→    queryKey: ["event", eventId],
    81→    queryFn: () => apiGet<EventDetailResponse>(`/api/events/${eventId}`).then((r) => r.data),
    82→    enabled: !!eventId,
    83→  });
    84→
    85→  const invalidate = () => {
    86→    qc.invalidateQueries({ queryKey: ["event", eventId] });
    87→    qc.invalidateQueries({ queryKey: ["events"] });
    88→    qc.invalidateQueries({ queryKey: ["overview"] });
    89→    qc.invalidateQueries({ queryKey: ["risk"] });
    90→  };
    91→
    92→  const reassess = useMutation({
    93→    mutationFn: () => apiPost<{ event: { riskScore: number; riskBand: string }; advisory: { provider: string } | null }>(`/api/events/${eventId}/reassess`, { useProvider: true }, "ANALYST"),
    94→    onSuccess: (r) => {
    95→      invalidate();
    96→      toast({ title: "Risk reassessed", description: `New score ${r.data.event.riskScore} (${r.data.event.riskBand}). Advisory provider: ${r.data.advisory?.provider ?? "rule engine"}.` });
    97→    },
    98→    onError: (e) => toast({ title: "Reassessment failed", description: (e as ApiClientError).message, variant: "destructive" }),
    99→  });
   100→
   101→  if (!eventId) {
   102→    return (
   103→      <div className="flex-1 grid place-items-center">
   104→        <EmptyState icon={<FileWarning className="size-8" />} title="No event selected" hint="Open an event from the command center or map." action={
   105→          <Button variant="outline" size="sm" onClick={() => navigate("command")} className="mt-2">← Command Center</Button>
   106→        } />
   107→      </div>
   108→    );
   109→  }
   110→  if (detailQ.isLoading) return <div className="p-6"><LoadingRows rows={8} /></div>;
   111→  if (detailQ.isError)
   112→    return (
   113→      <div className="p-6 max-w-lg">
   114→        <ErrorNote message={(detailQ.error as Error).message} onRetry={() => detailQ.refetch()} />

   116→      </div>
   117→    );
   118→
   119→  if (!detailQ.data) return null;
   120→  const { event, context, riskHistory, auditHistory } = detailQ.data;

   130→            <ArrowLeft className="size-4" />
   131→          </button>
   132→          <div className="min-w-0 flex-1">
   133→            <div className="flex items-center gap-2 flex-wrap">

   234→function QuickFacts({ d }: { d: D }) {
   235→  const e = d.event;
   236→  return (
   237→    <div className="p-4 space-y-4 xl:overflow-y-auto">

   240→        <RiskMeter score={e.riskScore} band={e.riskBand} />
   241→        <p className="mt-1 micro-label !text-[0.52rem] text-muted-foreground/70">
   242→          {e.riskModelVersion} · assessed {e.riskAssessedAt ? timeAgo(e.riskAssessedAt) : "-"}
   243→        </p>
   244→      </div>
   245→      <dl className="space-y-2 text-xs">
   246→        <Fact label="severity" value={<SeverityTicks severity={e.severity} />} />
   247→        <Fact label="reports" value={<span className="data-mono">{e.reportCount} ({e.reports.filter((r) => r.isDuplicate).length} merged)</span>} />
   248→        <Fact label="recurrence" value={<span className="data-mono">{e.recurrenceCount}×</span>} />
   249→        <Fact label="rainfall 24h/72h" value={<span className="data-mono text-water">{Math.round(e.rainfall24hMm ?? 0)} / {Math.round(e.rainfall72hMm ?? 0)} mm</span>} />
   250→        <Fact label="category" value={<span className="data-mono">{e.category.toLowerCase()}</span>} />
   251→        <Fact label="classification" value={<ProviderChip provider={e.classificationProvider ?? "-"} model={e.modelVersion} />} />
   252→        {e.groundTruthHotspotId && (
   253→          <Fact label="ground truth" value={<span className="text-[0.65rem] text-sev-moderate/80">{e.groundTruthHotspotId} · {e.groundTruthAgencyCode} (synthetic)</span>} />
   254→        )}
   255→      </dl>
   256→    </div>
   257→  );
   258→}

   266→  );

   271→function OverviewTab({ d }: { d: D }) {
   272→  const { event: e, context } = d;
   273→  const nearbyAssets = context.assets
   274→    .map((a) => ({ ...a, d: Math.round(hav({ lat: e.lat, lng: e.lng }, a)) }))
   275→    .filter((a) => a.d <= 600)
   276→    .sort((a, b) => a.d - b.d)
   277→    .slice(0, 6);
   278→  const nearIncidents = context.historicalIncidents
   279→    .map((i) => ({ ...i, d: Math.round(hav({ lat: e.lat, lng: e.lng }, i)) }))
   280→    .filter((i) => i.d <= 300)
   281→    .sort((a, b) => b.occurredOn.localeCompare(a.occurredOn))
   282→    .slice(0, 5);
   283→  const nearMaint = context.maintenance
   284→    .filter((m) => m.asset && nearbyAssets.some((a) => a.code === m.asset?.code))
   285→    .slice(0, 4);
   286→
   287→  return (
   288→    <>
   289→      <Panel title={`citizen reports · ${e.reportCount}`}>
   290→        <ul className="divide-y divide-border/60">
   291→          {e.reports.map((r) => (
   292→            <li key={r.id} className="py-2.5 first:pt-0 last:pb-0">
   293→              <div className="flex items-center gap-2 flex-wrap">
   294→                <span className="data-mono text-[0.65rem] text-water/90">{r.publicRef}</span>
   295→                <span className="micro-label !text-[0.52rem]">{r.channel.toLowerCase()}</span>
   296→                {r.isDuplicate && <span className="micro-label !text-[0.52rem] text-sev-moderate/80">merged as duplicate</span>}
   297→                <TimeAgo iso={r.submittedAt} />
   298→              </div>
   299→              <p className="mt-1 text-[0.82rem] text-foreground/90 leading-relaxed">{r.description}</p>
   300→              {r.classification && (
   301→                <div className="mt-1.5 flex items-center gap-2 flex-wrap text-[0.65rem] text-muted-foreground">
   302→                  <span>assessed <span className={cn("font-medium", sevClass(r.classification.severity))}>{r.classification.severity.toLowerCase()}</span></span>
   303→                  <span className="data-mono">conf {Math.round(r.classification.confidence * 100)}%</span>
   304→                  {r.classificationProvider && <ProviderChip provider={r.classificationProvider} />}
   305→                </div>
   306→              )}
   307→            </li>
   308→          ))}
   309→        </ul>
   310→      </Panel>
   311→
   312→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   313→        <Panel title={`infrastructure context · within 600m`}>
   314→          {nearbyAssets.length === 0 ? (
   315→            <EmptyState title="No assets recorded nearby" hint="Asset registry covers pilot jurisdictions only (synthetic)." />
   316→          ) : (
   317→            <ul className="space-y-2">
   318→              {nearbyAssets.map((a) => (
   319→                <li key={a.code} className="flex items-center gap-2.5 text-xs">
   320→                  <span className="data-mono text-[0.62rem] text-muted-foreground w-12 shrink-0">{a.d}m</span>
   321→                  <span className={cn("size-2 rounded-full shrink-0", a.conditionScore < 45 ? "bg-sev-high" : a.conditionScore < 65 ? "bg-sev-moderate" : "bg-verified")} aria-hidden />
   322→                  <span className="flex-1 min-w-0">
   323→                    <span className="block truncate">{a.name}</span>
   324→                    <span className="micro-label !text-[0.5rem]">{a.kind.toLowerCase().replace(/_/g, " ")} · {a.agencyCode} · condition {a.conditionScore}</span>
   325→                  </span>
   326→                </li>
   327→              ))}
   328→            </ul>
   329→          )}
   330→        </Panel>
   331→
   332→        <Panel title="historical incidents · within 300m">
   333→          {nearIncidents.length === 0 ? (
   334→            <EmptyState title="No historical incidents nearby" hint="Synthetic archive covers the pilot jurisdictions." />
   335→          ) : (
   336→            <ul className="space-y-2">
   337→              {nearIncidents.map((i) => (
   338→                <li key={i.id} className="flex items-baseline gap-2.5 text-xs">
   339→                  <span className="data-mono text-[0.62rem] text-muted-foreground shrink-0 w-20">{new Date(i.occurredOn).toLocaleDateString("en-IN", { month: "short", year: "numeric" })}</span>
   340→                  <span className={cn("font-medium shrink-0", sevClass(i.severity))}>{i.severity.toLowerCase()}</span>
   341→                  <span className="text-muted-foreground truncate">{i.waterDepthCm ?? "?"}cm · {i.durationHours?.toFixed(0) ?? "?"}h · {i.reportedVia.toLowerCase()}</span>
   342→                </li>
   343→              ))}
   344→            </ul>
   345→          )}
   346→        </Panel>
   347→      </div>
   348→
   349→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   350→        <Panel title="maintenance nearby">
   351→          {nearMaint.length === 0 ? (
   352→            <EmptyState title="No maintenance records for nearby assets" />
   353→          ) : (
   354→            <ul className="space-y-2">
   355→              {nearMaint.map((m, i) => (
   356→                <li key={i} className="flex items-baseline gap-2.5 text-xs">
   357→                  <span className="data-mono text-[0.62rem] text-muted-foreground w-20 shrink-0">
   358→                    {m.performedAt ? new Date(m.performedAt).toLocaleDateString("en-IN", { month: "short", year: "numeric" }) : "scheduled"}
   359→                  </span>
   360→                  <span className="flex-1 truncate">{m.asset?.name ?? "asset"} · {m.kind.toLowerCase()}</span>
   361→                  <span className={cn("micro-label !text-[0.5rem]", m.status === "COMPLETED" ? "text-verified" : "text-water")}>{m.status.toLowerCase()}</span>
   362→                </li>
   363→              ))}
   364→            </ul>
   365→          )}
   366→        </Panel>
   367→
   368→        <Panel title={<span className="flex items-center gap-1.5"><Umbrella className="size-3 text-water" aria-hidden /> rainfall · 72h (synthetic)</span>}>
   369→          <ul className="space-y-1.5 text-xs">
   370→            {context.rainfallSeries.slice(-12).map((s, i) => (
   371→              <li key={i} className="flex items-center gap-2">
   372→                <span className="data-mono text-[0.6rem] text-muted-foreground w-11 shrink-0">{new Date(s.t).toLocaleTimeString("en-IN", { hour: "2-digit", hour12: false })}</span>
   373→                <span className="h-1.5 rounded-full bg-water/70" style={{ width: `${Math.min(100, s.mm * 3)}%` }} />
   374→                <span className="data-mono text-[0.62rem]">{s.mm}mm</span>
   375→              </li>
   376→            ))}
   377→          </ul>
   378→        </Panel>
   379→      </div>
   380→    </>
   381→  );
   382→}

   394→

   397→      apiPost(`/api/events/${e.id}/verify`, { stage: "EVIDENCE_UPLOADED", notes: payload.caption, photoDataUrl: kind === "FIELD_PHOTO" ? payload.content : undefined, photoCaption: payload.caption }, "FIELD_TEAM"),

   432→                    <div className="size-full p-2.5 overflow-auto">
   433→                      <p className="text-[0.6rem] data-mono text-muted-foreground break-words">{ev.content.slice(0, 260)}</p>

   642→function ResponseTab({ d, onMutated }: { d: D; onMutated: () => void }) {
   643→  const { event: e } = d;
   644→  const { toast } = useToast();
   645→  const qc = useQueryClient();
   646→
   647→  const actionUpdate = useMutation({
   648→    mutationFn: (p: { actionId: string; status: string; outcome?: string }) => apiPut(`/api/events/${e.id}/actions`, p, "AGENCY"),
   649→    onSuccess: () => {
   650→      onMutated();
   651→      toast({ title: "Action updated" });
   652→    },
   653→    onError: (err) => toast({ title: "Update failed", description: (err as ApiClientError).message, variant: "destructive" }),
   654→  });
   655→
   656→  const verify = useMutation({
   657→    mutationFn: (p: Record<string, unknown>) => apiPost(`/api/events/${e.id}/verify`, p, "FIELD_TEAM"),
   658→    onSuccess: () => {
   659→      onMutated();
   660→      qc.invalidateQueries({ queryKey: ["overview"] });
   661→      toast({ title: "Verification stage recorded" });
   662→    },
   663→    onError: (err) => toast({ title: "Verification failed", description: (err as ApiClientError).message, variant: "destructive" }),
   664→  });
   665→
   666→  const reopen = useMutation({
   667→    mutationFn: (p: { reopenReason: string }) => apiPut(`/api/events/${e.id}/verify`, p, "AGENCY"),
   668→    onSuccess: () => {
   669→      onMutated();
   670→      toast({ title: "Event reopened", description: "Recurrence watch incremented." });
   671→    },
   672→    onError: (err) => toast({ title: "Reopen failed", description: (err as ApiClientError).message, variant: "destructive" }),
   673→  });
   674→
   675→  const assign = useMutation({
   676→    mutationFn: (p: { agencyCode: string; role: string; note?: string }) => apiPost(`/api/events/${e.id}/assign`, p, "ANALYST"),
   677→    onSuccess: () => {
   678→      onMutated();
   679→      toast({ title: "Responsibility assigned" });
   680→    },
   681→    onError: (err) => toast({ title: "Assignment failed", description: (err as ApiClientError).message, variant: "destructive" }),
   682→  });
   683→
   684→  return (
   685→    <>
   686→      <Panel title="responsibility chain" actions={<span className="micro-label !text-[0.5rem] text-muted-foreground/60">routing-rules v1.2</span>}>
   687→        <ol className="space-y-0">
   688→          <ChainLevel level="EVENT" label={e.code} detail={`${e.reportCount} reports · risk ${e.riskScore} (${e.riskBand})`} />
   689→          <ChainLevel level="JURISDICTION" label={e.jurisdiction?.name ?? "-"} detail={e.jurisdiction ? `${e.jurisdiction.kind.toLowerCase()} · administered by ${e.jurisdiction.agencyCode ?? "n/a"}` : "outside pilot"} />
   690→          {e.links.map((l) => (
   691→            <ChainLevel
   692→              key={l.id}
   693→              level={l.role}
   694→              label={l.agencyCode}
   695→              detail={l.reason ?? ""}
   696→              asset={l.asset ? `${l.asset.name} · condition ${l.asset.conditionScore}` : null}
   697→              status={l.status}
   698→              source={l.source}
   699→            />
   700→          ))}
   701→        </ol>
   702→        <AssignForm onAssign={(p) => assign.mutateAsync(p)} pending={assign.isPending} />
   703→        {e.groundTruthAgencyCode && (
   704→          <p className="mt-3 text-[0.65rem] text-sev-moderate/80">
   705→            Routing evaluation note: seeded ground truth for this location is <span className="data-mono">{e.groundTruthAgencyCode}</span> (synthetic label for research metrics).
   706→          </p>
   707→        )}
   708→      </Panel>
   709→
   710→      <Panel title={`action items · ${e.actions.length}`}>
   711→        {e.actions.length === 0 ? (
   712→          <EmptyState title="No action items" hint="Actions are recommended by the routing engine at triage." />
   713→        ) : (
   714→          <ul className="space-y-2">
   715→            {e.actions.map((a) => (
   716→              <li key={a.id} className="rounded-sm border border-border/60 bg-ink-850/30 p-3">
   717→                <div className="flex items-center gap-2 flex-wrap">
   718→                  <span className="data-mono text-[0.62rem] font-semibold text-water">{a.kind.replace(/_/g, " ").toLowerCase()}</span>
   719→                  <span className={cn("micro-label !text-[0.5rem]", a.priority === "URGENT" ? "text-sev-critical" : a.priority === "HIGH" ? "text-sev-high" : "text-muted-foreground")}>{a.priority.toLowerCase()}</span>
   720→                  <span className="ml-auto data-mono text-[0.62rem] text-muted-foreground">{a.agencyCode ?? "-"}{a.assignedTo ? ` · ${a.assignedTo}` : ""}</span>

   723→                {a.outcome && <p className="mt-1 text-xs text-verified/90">Outcome: {a.outcome}</p>}

   731→                    <div className="ml-auto flex gap-1.5">
   732→                      {a.status === "RECOMMENDED" && (
   733→                        <Button size="sm" variant="outline" className="h-7 text-[0.65rem] border-border" onClick={() => actionUpdate.mutate({ actionId: a.id, status: "ASSIGNED" })}>
   734→                          accept
   735→                        </Button>
   736→                      )}
   737→                      <Button size="sm" variant="outline" className="h-7 text-[0.65rem] border-border" onClick={() => actionUpdate.mutate({ actionId: a.id, status: "IN_PROGRESS" })}>
   738→                        start
   739→                      </Button>
   740→                      <Button size="sm" className="h-7 text-[0.65rem] bg-verified/85 text-ink-950 hover:bg-verified" onClick={() => actionUpdate.mutate({ actionId: a.id, status: "COMPLETED", outcome: "Completed via event dossier." })}>
   741→                        complete

   743→                    </div>
   744→                  )}
   745→                </div>
   746→              </li>
   747→            ))}
   748→          </ul>
   749→        )}
   750→      </Panel>
   751→
   752→      <Panel title="field verification workflow">
   753→        <VerificationStepper stages={e.verifications.map((v) => v.stage)} />
   754→        <VerificationTimeline verifications={e.verifications} />
   755→        <VerificationForms
   756→          event={e}
   757→          onVerify={(p) => verify.mutateAsync(p)}
   758→          onReopen={(reason) => reopen.mutateAsync({ reopenReason: reason })}
   759→          pending={verify.isPending}
   760→        />
   761→      </Panel>
   762→    </>
   763→  );

  1006→function hav(a: { lat: number; lng: number }, b: { lat: number; lng: number }): number {
  1007→  const R = 6371000;
  1008→  const rad = (x: number) => (x * Math.PI) / 180;
  1009→  const dLat = rad(b.lat - a.lat);
  1010→  const dLng = rad(b.lng - a.lng);
  1011→  const s = Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLng / 2) ** 2;
  1012→  return 2 * R * Math.asin(Math.sqrt(s));
  1013→}

  1015→async function fileToDataUrl(file: File, maxDim: number, quality: number): Promise<string> {
  1016→  const bitmap = await createImageBitmap(file);
  1017→  const scale = Math.min(1, maxDim / Math.max(bitmap.width, bitmap.height));
  1018→  const canvas = document.createElement("canvas");
  1019→  canvas.width = Math.round(bitmap.width * scale);
  1020→  canvas.height = Math.round(bitmap.height * scale);
  1021→  const ctx = canvas.getContext("2d");
  1022→  if (!ctx) throw new Error("Canvas unavailable");
  1023→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  1024→  bitmap.close();
  1025→  return canvas.toDataURL("image/jpeg", quality);
  1026→}
